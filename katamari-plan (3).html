Coding style we copy from the labs

These are the conventions visible in all three lab samples. Following them keeps the project recognisably “ours from the course”. Where the game needs something no lab shows, the phase card says so with not in labs and uses the smallest step away from these conventions.

Do it like the labs

  • Header comment on every file (course, year, authors) and "use strict"; with the labs' explanatory comment.
  • Globals at the top with var (canvas, gl, matrices, locations); constants as const UPPER_CASE like ANGLE_INCREMENT, each with a comment giving its meaning and unit.
  • Handlers assigned directly: window.onload = init;, window.onkeydown = onKeyDown;, named onSomething(e).
  • init() with console.log("init... Begin")/"… End"; WebGL calls in the Lab 08 order, one short comment above each.
  • transform() computes matrices, render() draws (Lab 10/Lab 12). Matrices named matrixProjection, matrixModelview.
  • Non-indexed triangle lists: arrays of vec4 pushed triangle by triangle, uploaded with bufferData(…, flatten(…), gl.STATIC_DRAW) and drawn with drawArrays(gl.TRIANGLES, 0, n). No element buffers (no lab uses them).
  • Shaders inline in the HTML, precision mediump float;, names aSomething (attribute), uSomething (uniform), vSomething (varying) as in Lab 12. JavaScript location variables use the same names (aPosition, uImage0).
  • Classes as in Lab 10/Lab 12: fields listed at the top with comments, set in the constructor, private helpers as #method, .bind(this) when a method is used as a callback.
  • HTML: grey #888888 body, <center>, canvas gl-canvas, a <div> line explaining the controls (Lab 10/Lab 12: “Press +/- for scaling…”).

Lab quirks we must not copy

  • class Object (Lab 10, Lab 12) hides JavaScript's built-in Object. Name our classes Mesh, Sphere, Player, Camera.
  • gl.enable(gl.TEXTURE_2D) (Lab 12 render()) is desktop OpenGL, not WebGL: Chromium logs INVALID_ENUM: enable: invalid capability every frame (checked). Leave it out.
  • perspective(90, 2, 0.1, 10) in Lab 12 hard-codes the aspect ratio and stretches the image; use canvas.width / canvas.height as Lab 10 does.
  • Lab 12 folds the camera translate into matrixProjection. That breaks lighting, which needs the camera in the model-view; we use lookAt for the view (Phase 4).
  • Spelling slips such as onTexImageLaod and texParatmeteri: spell our names correctly, readability is 10% of the grade.
  • Lab 08 animates with setInterval(render, 50); for a game use requestAnimFrame from webgl-utils.js and a measured dt.
  • The starter header says “Fall 2025”: fix it to Fall 2026 with our names.

World conventions

  • Units are metres and seconds. Y is up, the ground is the plane y = 0, the sky is centred at the origin.
  • Every ball is the same unit sphere mesh, scaled by its radius in its model matrix. The sky is that sphere with radius SKY_RADIUS.
  • CULL_FACE stays off (the labs' default), so triangle winding never matters, including for the sky seen from inside.
  • Every speed and angle change is multiplied by dt (seconds since last frame), so the game runs the same at 30 or 144 fps.

<script id="vertex-shader" type="x-shader/x-vertex">
    attribute vec4 aPosition;
    attribute vec4 aColor;

    // Model-view and projection matrices (set from JavaScript every frame)
    uniform mat4 uModelView;
    uniform mat4 uProjection;

    // Varying variable to send the color to the fragment shader
    varying vec4 vColor;

    void main() {
        // Pass the color to the fragment shader
        vColor = aColor;

        // Transform the vertex: object -> camera -> clip coordinates
        gl_Position = uProjection * uModelView * aPosition;
    }
</script>

<script id="fragment-shader" type="x-shader/x-fragment">
    // Set how many bits used for float values, it won't work otherwise
    precision mediump float;

    // Varying variable to get color from the vertex shader
    varying vec4 vColor;

    void main() {
        gl_FragColor = vColor;
    }
</script>

function init() {
    console.log("init... Begin");
    // Get reference to the WebGL context of the canvas
    canvas = document.getElementById("gl-canvas");

    gl = WebGLUtils.setupWebGL(canvas);
    if (!gl) { alert("WebGL isn't available"); }

    //  Configure WebGL
    // Set the color value used when clearing color buffers.
    gl.clearColor(0.0, 0.0, 0.0, 1.0);
    // Hidden surface removal is always on in the game.
    gl.enable(gl.DEPTH_TEST);

    //  Load shaders and initialize attribute buffers
    program = initShaders(gl, "vertex-shader", "fragment-shader");
    gl.useProgram(program);

    aPosition   = gl.getAttribLocation(program, "aPosition");
    aColor      = gl.getAttribLocation(program, "aColor");
    uModelView  = gl.getUniformLocation(program, "uModelView");
    uProjection = gl.getUniformLocation(program, "uProjection");

    // ... copy the tetrahedron arrays and the createBuffer / bindBuffer / bufferData /
    //     vertexAttribPointer / enableVertexAttribArray calls from init() in gl-rotation lab ...

    lastTime = Date.now();
    render();
    console.log("init... End");
}

function transform() {
    // Projection: use the canvas aspect ratio to prevent stretching (gl-rotation lab comment)
    matrixProjection = perspective(60, canvas.width / canvas.height, 0.1, 100.0);
    // Camera: a fixed test camera for now (Phase 4 replaces it)
    matrixView = lookAt(vec3(0.0, 2.0, 5.0), vec3(0.0, 0.0, 0.0), vec3(0.0, 1.0, 0.0));
}

function render() {
    // Time step in seconds since the previous frame.
    // Capped at 0.05 s so the game does not jump after the tab was hidden.
    var now = Date.now();
    var dt = Math.min((now - lastTime) / 1000.0, 0.05);
    lastTime = now;

    // Clear color and depth buffers to preset values.
    gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);

    transform();

    // Test animation: turn the tetrahedron 90 degrees per second
    testAngle += 90.0 * dt;
    matrixModelview = mult(matrixView, rotateY(testAngle));

    // Specify the values of the uniform variables
    gl.uniformMatrix4fv(uProjection, false, flatten(matrixProjection));
    gl.uniformMatrix4fv(uModelView, false, flatten(matrixModelview));

    // Render primitives from the array data
    gl.drawArrays(gl.TRIANGLES, 0, triangleVertices.length);

    // Ask the browser to call render() again before the next repaint (common/webgl-utils.js)
    requestAnimFrame(render);
}

// Mesh class to store the triangles of one shape and their WebGL buffers.
// The game needs only two meshes: a unit sphere (used for every ball and the sky) and the ground plane.
class Mesh {
    // Arrays for storing all triangle data (like triangleVertices / triangleTexCoords in the labs)
    vertices;    // vec4 positions
    normals;     // vec3 unit normals, one per vertex
    texCoords;   // vec2 texture coordinates, one per vertex
    // WebGL buffer ids
    bufferVertexId;
    bufferNormalId;
    bufferTexCoordId;

    constructor() {
        this.vertices = new Array();
        this.normals = new Array();
        this.texCoords = new Array();
    }

    // Build a polyhedron that approximates a sphere of radius 1 centred at the origin.
    // stacks: number of horizontal bands (latitude), slices: number of cuts around the y-axis (longitude).
    makeSphere(stacks, slices) {
        for (var i = 0; i < stacks; i++) {
            for (var j = 0; j < slices; j++) {
                // Two triangles for the small quad between latitudes i, i+1 and longitudes j, j+1
                this.#addSpherePoint(i,     j,     stacks, slices);
                this.#addSpherePoint(i + 1, j,     stacks, slices);
                this.#addSpherePoint(i + 1, j + 1, stacks, slices);
                this.#addSpherePoint(i,     j,     stacks, slices);
                this.#addSpherePoint(i + 1, j + 1, stacks, slices);
                this.#addSpherePoint(i,     j + 1, stacks, slices);
            }
        }
    }

    #addSpherePoint(i, j, stacks, slices) {
        var theta = i * Math.PI / stacks;         // 0 at the north pole, PI at the south pole
        var phi = j * 2.0 * Math.PI / slices;     // angle around the y-axis, 0 to 2 PI
        var x = Math.sin(theta) * Math.cos(phi);
        var y = Math.cos(theta);
        var z = Math.sin(theta) * Math.sin(phi);
        this.vertices.push(vec4(x, y, z, 1.0));
        this.normals.push(vec3(x, y, z));          // on a unit sphere the normal equals the position
        this.texCoords.push(vec2(j / slices, i / stacks));
    }

    // Build a square at y = 0 from -halfSize to +halfSize. The texture repeats `tiles` times per side.
    makePlane(halfSize, tiles) {
        var corners = [
            vec4(-halfSize, 0.0, -halfSize, 1.0),
            vec4(-halfSize, 0.0,  halfSize, 1.0),
            vec4( halfSize, 0.0,  halfSize, 1.0),
            vec4( halfSize, 0.0, -halfSize, 1.0)
        ];
        var uvs = [vec2(0.0, 0.0), vec2(0.0, tiles), vec2(tiles, tiles), vec2(tiles, 0.0)];
        // Two triangles, listed corner by corner as in the labs' triangleVertices arrays
        var order = [0, 1, 2, 0, 2, 3];
        for (var k = 0; k < order.length; k++) {
            this.vertices.push(corners[order[k]]);
            this.normals.push(vec3(0.0, 1.0, 0.0));
            this.texCoords.push(uvs[order[k]]);
        }
    }

    // Load the triangle data into the GPU (same calls as in init() of the labs)
    loadToGPU() {
        this.bufferVertexId = gl.createBuffer();
        gl.bindBuffer(gl.ARRAY_BUFFER, this.bufferVertexId);
        gl.bufferData(gl.ARRAY_BUFFER, flatten(this.vertices), gl.STATIC_DRAW);

        this.bufferNormalId = gl.createBuffer();
        gl.bindBuffer(gl.ARRAY_BUFFER, this.bufferNormalId);
        gl.bufferData(gl.ARRAY_BUFFER, flatten(this.normals), gl.STATIC_DRAW);

        this.bufferTexCoordId = gl.createBuffer();
        gl.bindBuffer(gl.ARRAY_BUFFER, this.bufferTexCoordId);
        gl.bufferData(gl.ARRAY_BUFFER, flatten(this.texCoords), gl.STATIC_DRAW);
    }

    // Connect this mesh's buffers to the shader attributes and draw its triangles.
    // Attribute pointers are global WebGL state, so they must be set again for every mesh.
    draw() {
        gl.bindBuffer(gl.ARRAY_BUFFER, this.bufferVertexId);
        gl.vertexAttribPointer(aPosition, 4, gl.FLOAT, false, 0, 0);
        gl.enableVertexAttribArray(aPosition);

        gl.bindBuffer(gl.ARRAY_BUFFER, this.bufferNormalId);
        gl.vertexAttribPointer(aNormal, 3, gl.FLOAT, false, 0, 0);
        gl.enableVertexAttribArray(aNormal);

        // The GLSL compiler removes unused attributes and then getAttribLocation returns -1
        if (aTexCoord != -1) {
            gl.bindBuffer(gl.ARRAY_BUFFER, this.bufferTexCoordId);
            gl.vertexAttribPointer(aTexCoord, 2, gl.FLOAT, false, 0, 0);
            gl.enableVertexAttribArray(aTexCoord);
        }

        gl.drawArrays(gl.TRIANGLES, 0, this.vertices.length);
    }
}

// Sphere class for every ball in the scene: the free spheres, the attachments and the sky.
class Sphere {
    position;   // vec3 centre in world coordinates (metres)
    radius;     // radius in metres
    // Added in Phase 7 and 8: materialAmbient, materialDiffuse, materialSpecular, shininess, texture

    constructor(position, radius) {
        this.position = position;
        this.radius = radius;
    }

    // Model matrix: scale the unit sphere to the radius, then move it to the position
    getModelMatrix() {
        return mult(translate(this.position), scalem(this.radius, this.radius, this.radius));
    }

    // True if this sphere overlaps a sphere at otherPosition with otherRadius
    intersects(otherPosition, otherRadius) {
        return length(subtract(this.position, otherPosition)) < this.radius + otherRadius;
    }

    // Draw with the shared unit-sphere mesh. modelMatrix is optional (attachments pass their own).
    draw(modelMatrix) {
        if (modelMatrix === undefined) {
            modelMatrix = this.getModelMatrix();
        }
        setModelView(modelMatrix);
        sphereMesh.draw();
    }
}

// Place SPHERE_COUNT spheres at random positions on the plane so that no two of them intersect.
// Rejection sampling: pick a random position and radius, throw it away if it overlaps.
function placeSpheres() {
    spheres = [];
    var attempts = 0;
    while (spheres.length < SPHERE_COUNT && attempts < 10000) {
        attempts++;
        var r = SPHERE_MIN_RADIUS + Math.random() * (SPHERE_MAX_RADIUS - SPHERE_MIN_RADIUS);
        var angle = Math.random() * 2.0 * Math.PI;
        // sqrt() spreads the points evenly over the disc instead of crowding the centre
        var dist = Math.sqrt(Math.random()) * (SKY_RADIUS - PLAY_MARGIN - r);
        // y = r: the sphere rests on the plane
        var candidate = new Sphere(vec3(dist * Math.cos(angle), r, dist * Math.sin(angle)), r);

        var free = !candidate.intersects(vec3(0.0, PLAYER_RADIUS, 0.0), PLAYER_RADIUS + START_GAP);
        for (var k = 0; free && k < spheres.length; k++) {
            if (candidate.intersects(spheres[k].position, spheres[k].radius)) {
                free = false;
            }
        }
        if (free) {
            spheres.push(candidate);
        }
    }
    console.log("placeSpheres: " + spheres.length + " spheres after " + attempts + " attempts");
}

// Fields are added phase by phase; every one is listed here with its unit.
class Player extends Sphere {
    velocity;      // vec3, metres per second (Phase 6)
    orientation;   // mat4, rotation accumulated while rolling (Phase 10)
    attachments;   // array of { sphere, localOffset, localRotation } (Phase 11)
    totalVolume;   // cubic metres: core plus all attachments (Phase 11)

    constructor(position, radius) {
        super(position, radius);
        this.velocity = vec3(0.0, 0.0, 0.0);
        this.orientation = mat4();
        this.attachments = new Array();
        this.totalVolume = 4.0 / 3.0 * Math.PI * Math.pow(radius, 3);
    }

    // Radius of a single ball with the same total volume.
    // Used for the camera distance, the rolling speed and the "smaller / larger" test.
    effectiveRadius() {
        return Math.cbrt(3.0 * this.totalVolume / (4.0 * Math.PI));
    }
}

const CAMERA_HEIGHT = 4.0;         // how high above the ball

// Camera class: third-person camera that stays behind and above the player
class Camera {
    yaw;             // degrees around the y-axis; 0 looks along -z (changed by the mouse in Phase 5)
    heightFactor;    // height above the ball, in ball radii (changed by the mouse in Phase 5)
    distanceFactor;  // distance behind the ball, in ball radii
    aheadFactor;     // look-at point in front of the ball, in ball radii

    constructor() {
        this.yaw = 0.0;
        this.heightFactor = CAMERA_HEIGHT;
        this.distanceFactor = CAMERA_DISTANCE;
        this.aheadFactor = CAMERA_AHEAD;
    }

    // Unit vector pointing forward along the ground (MV.js trigonometry needs radians())
    getForward() {
        return vec3(Math.sin(radians(this.yaw)), 0.0, -Math.cos(radians(this.yaw)));
    }

    // Unit vector pointing to the right of forward, along the ground
    getRight() {
        return vec3(Math.cos(radians(this.yaw)), 0.0, Math.sin(radians(this.yaw)));
    }

    // View matrix that keeps the target at the bottom centre of the screen
    getViewMatrix(target, radius) {
        var forward = this.getForward();
        var eye = add(target, add(scale(-this.distanceFactor * radius, forward),
                                  vec3(0.0, this.heightFactor * radius, 0.0)));
        var at = add(target, scale(this.aheadFactor * radius, forward));
        return lookAt(eye, at, vec3(0.0, 1.0, 0.0));
    }
}

<script id="vertex-shader" type="x-shader/x-vertex">
    attribute vec4 aPosition;
    attribute vec3 aNormal;
    attribute vec2 aTexCoord;

    uniform mat4 uModelView;      // view * model
    uniform mat4 uProjection;
    uniform mat3 uNormalMatrix;   // normalMatrix(modelView, true) from MV.js
    uniform vec4 uLightPosition;  // light position already in camera (eye) coordinates

    // Vectors for the lighting calculation, interpolated across each triangle
    varying vec3 vNormal;
    varying vec3 vLight;
    varying vec3 vEye;
    varying vec2 vTexCoord;

    void main() {
        // Vertex position in eye coordinates (the camera is at the origin there)
        vec3 pos = (uModelView * aPosition).xyz;
        vNormal = uNormalMatrix * aNormal;
        vLight = uLightPosition.xyz - pos;   // from the surface towards the light
        vEye = -pos;                         // from the surface towards the camera
        vTexCoord = aTexCoord;               // used in Phase 8
        gl_Position = uProjection * uModelView * aPosition;
    }
</script>

<script id="fragment-shader" type="x-shader/x-fragment">
    precision mediump float;

    varying vec3 vNormal;
    varying vec3 vLight;
    varying vec3 vEye;
    varying vec2 vTexCoord;

    // Light colour x material colour, computed in JavaScript with MV.js mult()
    uniform vec4 uAmbientProduct;
    uniform vec4 uDiffuseProduct;
    uniform vec4 uSpecularProduct;
    uniform float uShininess;     // Phong exponent: larger = smaller, sharper highlight

    void main() {
        vec3 N = normalize(vNormal);
        vec3 L = normalize(vLight);
        vec3 E = normalize(vEye);
        vec3 H = normalize(L + E);            // halfway vector (Blinn-Phong)

        float Kd = max(dot(L, N), 0.0);       // diffuse: surface facing the light is brighter
        float Ks = pow(max(dot(N, H), 0.0), uShininess);   // specular highlight
        if (dot(L, N) < 0.0) {
            Ks = 0.0;                         // no highlight on the dark side
        }

        vec4 color = uAmbientProduct + Kd * uDiffuseProduct + Ks * uSpecularProduct;
        color.a = 1.0;
        gl_FragColor = color;
    }
</script>

11Collisions: attach smaller, bounce off larger, grow B2

Each frame, test the player core and every attachment against every free sphere. A touched sphere smaller than the katamari is stuck on (stored in the katamari's local frame) and its volume is added; a larger one makes the katamari bounce.

Phases 3 (intersects), 4 (effectiveRadius), 6 (velocity), 10 (orientation).

Lab reference
Matrix composition as in transform() of Lab 10. not in labs collision detection and response; sphere–sphere distance tests and reflecting the velocity about the contact normal are the simplest standard methods.
common/
transpose, mult (mat4 × vec4 with w = 0), subtract, add, dot, normalize, scale, length, translate, scalem.
How
    // Stick sphere s to the katamari. The offset is stored in the katamari's own frame,
    // so it turns with the ball. For a pure rotation matrix, transpose() is the inverse.
    attach(s) {
        var inverse = transpose(this.orientation);
        var offset = mult(inverse, vec4(subtract(s.position, this.position), 0.0));
        this.attachments.push({
            sphere: s,
            localOffset: vec3(offset[0], offset[1], offset[2]),
            localRotation: inverse      // keeps the texture from jumping at the moment of attaching
        });
        // The katamari grows by the volume of the new sphere
        this.totalVolume += 4.0 / 3.0 * Math.PI * Math.pow(s.radius, 3);
    }

    // World position of one attachment (w = 0: rotate the offset, do not translate it)
    getAttachmentPosition(att) {
        var offset = mult(this.orientation, vec4(att.localOffset, 0.0));
        return add(this.position, vec3(offset[0], offset[1], offset[2]));
    }

    // Model matrix of one attachment: it rotates with the ball around the ball's centre
    getAttachmentModelMatrix(att) {
        var r = att.sphere.radius;
        return mult(translate(this.position), mult(this.orientation,
               mult(translate(att.localOffset), mult(att.localRotation, scalem(r, r, r)))));
    }

    // Core and attachments as { position, radius } in world coordinates (for collisions)
    getParts() {
        var parts = [{ position: this.position, radius: this.radius }];
        for (var k = 0; k < this.attachments.length; k++) {
            parts.push({ position: this.getAttachmentPosition(this.attachments[k]),
                         radius: this.attachments[k].sphere.radius });
        }
        return parts;
    }

    // Bounce off a larger sphere s that touches `part`: reflect the velocity about the
    // horizontal contact normal and push the katamari out of the overlap.
    bounceOff(part, s) {
        var n = subtract(part.position, s.position);
        n[1] = 0.0;                            // keep the motion on the ground
        if (length(n) == 0.0) {
            return;                            // normalize() of a zero vector gives NaN
        }
        n = normalize(n);
        var vn = dot(this.velocity, n);
        if (vn < 0.0) {                        // only if moving into the sphere
            this.velocity = subtract(this.velocity, scale((1.0 + RESTITUTION) * vn, n));
        }
        var overlap = s.radius + part.radius - length(subtract(part.position, s.position));
        this.position = add(this.position, scale(overlap + 0.01, n));
    }
const ATTACH_RATIO = 1.0;   // a sphere sticks if its radius < ATTACH_RATIO * player.effectiveRadius()
const RESTITUTION = 0.6;    // 0 = no bounce, 1 = perfectly elastic bounce

// Check the player core and every attachment against every free sphere (Requirement B2)
function checkCollisions() {
    var parts = player.getParts();
    // Walk backwards so splice() does not skip the next sphere
    for (var i = spheres.length - 1; i >= 0; i--) {
        var s = spheres[i];
        for (var k = 0; k < parts.length; k++) {
            if (s.intersects(parts[k].position, parts[k].radius)) {
                if (s.radius < ATTACH_RATIO * player.effectiveRadius()) {
                    player.attach(s);
                    spheres.splice(i, 1);      // no longer a free sphere
                } else {
                    player.bounceOff(parts[k], s);
                }
                break;
            }
        }
    }
}
